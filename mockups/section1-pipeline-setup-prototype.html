# Generated from the SSOT field-map. Same artifact validated on the harness.
from pyspark.sql.functions import from_json, col
from lakets import ChronoTable, ensure_partitions

BOOTSTRAP   = "b-1.iotdemo…:9098,b-2.iotdemo…:9098"
TOPIC       = "iot.pharmaceutical.realtime"
CREDENTIAL  = "iot-msk-cred"          # UC service credential
PARTITIONS  = 6

# SSOT parse schema (realtime) — cannot drift from the sink DDL
schema = sensor_record_schema()

# Hard-gate: 24h / 96-chunk forward window before any write
ensure_partitions(table="sensor_readings", horizon_hours=24, chunks=96)

df = (spark.readStream.format("kafka")
        .option("kafka.bootstrap.servers", BOOTSTRAP)
        .option("subscribe", TOPIC)
        .option("kafka.sasl.mechanism", "AWS_MSK_IAM")
        .option("startingOffsets", "latest")
        .load())

parsed = (df.select(from_json(col("value").cast("string"), schema).alias("r"))
          .select("r.*"))

(parsed.writeStream
    .trigger(realTime="5 minutes")                # real-time mode
    .foreachBatch(ChronoTable("databricks_postgres").upsert)
    .option("checkpointLocation", CHECKPOINT)
    .start())
